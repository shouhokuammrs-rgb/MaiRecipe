# 冷蔵庫と買い物の一本化（#38 PR1）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 冷蔵庫（pantry_items）を足し、買い物リストの「家にある」「買った」を「冷蔵庫にある（have）」1つのチェックにまとめ、今の印を冷蔵庫へ移す。

**Architecture:** 新しい表 `pantry_items`（グループ×名寄せ後の名前で1行）。買い物リストは「調味料以外＝冷蔵庫に同じ名前があるか」「調味料＝『家にない』印が無いか」で `have` を返す。純粋な関数と zod は `src/shared/pantry.ts`、DB は `forGroup` に足し、API は `/api/pantry` と `/api/shopping/have`。画面は新しいタブ「冷蔵庫」と、買い物の1チェック化。

**Tech Stack:** Cloudflare Workers + Hono、D1 + Drizzle、React 19 + TanStack Query + Tailwind v4、Vitest（`@cloudflare/vitest-pool-workers`）、Playwright

**Spec:** `docs/superpowers/specs/2026-09-30-pantry-and-recommend-design.md`（この計画は §1〜§3 の PR1 部分と §5 の「冷蔵庫（PR1）」「買い物」を作る）

## Global Constraints

- グループの判定はセッションからだけ。DB は `src/api/data/` の `forGroup` 経由だけ。routes から DB を直接触らない（ESLint で禁止）
- 新しい表・API には `test/api` に「グループをまたいだ漏れがないこと」を足す（見えない・直せない・消せない＝404・買い物に効かない）
- D1 は1つのクエリに渡せる値が100個まで。複数行の INSERT は1行ずつの文にして `db.batch` で流す
- Worker の CPU は 10ms まで。重い処理をしない
- 検索条件・集計に SQLite 独自関数を使わない（マイグレーションの中だけは可）
- 画面: 日本語、375px 基準、タップ領域 44px 以上、ローディング／エラー／空状態を必ず出す。`confirm()` は使わない（消すのは「使い切った → 本当に消す？」の2回押し）。色はトークン（paper / ink / sub / faint / accent / herb / herb-mid / memo / line-soft / field / card / danger …。`src/web/index.css` の @theme を見て、そこにある名前だけ使う）
- 冷蔵庫と買い物の名前の照合は `canonicalName` 後の **完全一致**
- 調味料（`shopSection(canonicalName(name)) === "調味料"`）は冷蔵庫に入れない・一覧に出さない
- PR 前に `npm run lint && npm run typecheck && npm run test` と `npm run e2e`
- コミットの最後に `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`

## Review Focus

1. 冷蔵庫の追加で 30 個まとめて送ったとき、D1 の 100 個の値の上限に当たらない（1行ずつの INSERT を batch）→ Task 3 に 30 個のテスト
2. 「玉葱」を入れたら買い物の「玉ねぎ」が隠れる（名寄せしてから保存・照合）→ Task 3・4 のテスト
3. 調味料を冷蔵庫に入れようとしても入らず skipped に返る。買い物で調味料のチェックを外すと「家にない」になり、また付けると戻る → Task 3・4
4. 他のグループの冷蔵庫の id で PATCH / DELETE すると 404、他人の冷蔵庫は自分の買い物の have に効かない → Task 3・4
5. 移行で「家にない」と「買った」の両方が付いていた調味料は「家にある」に戻る。単位違いの「家にない」印が2つあっても移行が落ちない → Task 2

```ts
// src/shared/pantry.ts
// 冷蔵庫（棚卸し）の純粋な関数と入力の検証。AI は使わない。
import { z } from "zod";
import { addDays, isDate } from "./dates";
import { canonicalName, shopSection } from "./ingredients";

/** 期限が「近い」とみなす日数（今日から数えてこの日数以内。過ぎたものも近い） */
export const PANTRY_SOON_DAYS = 3;

/** 「玉ねぎ にんじん、豚こま」を名前の配列にする。名寄せして重複を消す */
export function splitPantryInput(text: string): string[] {
  const names = text
    .split(/[\s　、,，]+/)
    .map(canonicalName)
    .filter(Boolean);
  return [...new Set(names)];
}

/** 調味料はいつも家にある扱いで、冷蔵庫には入れない */
export function isSeasoning(name: string): boolean {
  return shopSection(canonicalName(name)) === "調味料";
}

function daysBetween(from: string, to: string): number {
  return Math.round(
    (Date.parse(`${to}T00:00:00Z`) - Date.parse(`${from}T00:00:00Z`)) /
      86_400_000,
  );
}

export function addedLabel(addedOn: string, today: string): string {
  const d = daysBetween(addedOn, today);
  if (d <= 0) return "今日入れた";
  if (d === 1) return "昨日入れた";
  return `${d}日前に入れた`;
}

export function isExpiringSoon(expiresOn: string | null, today: string) {
  return expiresOn !== null && expiresOn <= addDays(today, PANTRY_SOON_DAYS);
}

export function sortPantry<
  T extends { name: string; expiresOn: string | null; addedOn: string },
>(items: T[]): T[] {
  return [...items].sort((a, b) => {
    if (a.expiresOn !== b.expiresOn) {
      if (a.expiresOn === null) return 1;
      if (b.expiresOn === null) return -1;
      return a.expiresOn < b.expiresOn ? -1 : 1;
    }
    if (a.addedOn !== b.addedOn) return a.addedOn > b.addedOn ? -1 : 1;
    return a.name.localeCompare(b.name, "ja");
  });
}

const pantryName = z.string().trim().min(1).max(40);

export const pantryAddSchema = z.object({
  names: z.array(pantryName).min(1).max(30),
});

export const pantryPatchSchema = z
  .object({
    amount: z
      .string()
      .trim()
      .max(40)
      .nullable()
      .optional()
      .transform((v) => (v === "" ? null : v)),
    expiresOn: z
      .string()
      .refine(isDate, "日付を確認してください")
      .nullable()
      .optional(),
  })
  .refine((p) => p.amount !== undefined || p.expiresOn !== undefined, {
    message: "直す項目がありません",
  });

export const shoppingHaveSchema = z.object({
  name: pantryName,
  value: z.boolean(),
});
```

注意: `pantryPatchSchema.parse({ amount: "  " })` が `{ amount: null }` になり、`expiresOn` キーが出ないこと（zod の optional は無いキーを足さない）。テストが通らなければ transform の位置を直す。

```ts
// test/api/pantry-migration.test.ts
import { env } from "cloudflare:test";
import { describe, expect, it } from "vitest";
import { api, signUpAs } from "./helpers";

async function groupOf(email: string): Promise<string> {
  const row = await env.DB.prepare(
    `select gm.group_id as g from group_members gm join user u on u.id = gm.user_id where u.email = ?`,
  )
    .bind(email)
    .first<{ g: string }>();
  return row!.g;
}

/** 0004 のうち、表を作る文以外（＝データの移し替え）をもう一度流す */
async function runPantryDataMigration() {
  const m = env.TEST_MIGRATIONS.find((x) => x.name.startsWith("0004"));
  if (!m) throw new Error("0004 が無い");
  for (const q of m.queries) {
    if (/^\s*CREATE/i.test(q)) continue;
    await env.DB.prepare(q).run();
  }
}

describe("#38 の移行：買い物の印を冷蔵庫へ", () => {
  it("家にある・買ったは冷蔵庫へ、残るのは調味料の「家にない」だけ", async () => {
    const me = await signUpAs("移行");
    // ログインしただけではグループが無いことがあるので、API を1回呼んでグループを作らせる
    await api(me.cookie, "/shopping?days=1");
    const g = await groupOf(me.email);
    const mark = (key: string, kind: string, value: number) =>
      env.DB.prepare(
        `insert into shopping_marks (group_id, key, kind, value) values (?, ?, ?, ?)`,
      )
        .bind(g, key, kind, value)
        .run();
    await mark("玉ねぎ|個", "home", 1);
    await mark("玉ねぎ|g", "bought", 1); // 単位違いでも冷蔵庫では1行
    await mark("卵|個", "bought", 1);
    await mark("牛乳|ml", "bought", 0); // 付けて外した印は移さない
    await mark("醤油|大さじ", "home", 0); // 調味料の「家にない」→ 残る
    await mark("醤油|小さじ", "home", 0); // 単位違い → 1つにまとまる
    await mark("みりん|大さじ", "home", 0);
    await mark("みりん|大さじ", "bought", 1); // 買ったので家にある → 印は消える

    await runPantryDataMigration();

    const pantry = await env.DB.prepare(
      `select name, added_on from pantry_items where group_id = ? order by name`,

      .bind(g)
      .all<{ name: string; added_on: string }>();
    expect(pantry.results.map((r) => r.name).sort()).toEqual(
      ["みりん", "卵", "玉ねぎ"].sort(),

    expect(pantry.results[0]!.added_on).toMatch(/^\d{4}-\d{2}-\d{2}$/);

    const marks = await env.DB.prepare(
      `select key, kind, value from shopping_marks where group_id = ?`,
    )
      .bind(g)
      .all<{ key: string; kind: string; value: number }>();
    expect(marks.results).toEqual([{ key: "醤油", kind: "home", value: 0 }]);
  });
});
```

```ts
    async addPantry(
      names: string[],
      today: string,
    ): Promise<{ added: string[]; skipped: string[] }> {
      const skipped: string[] = [];
      const wanted: string[] = [];
      for (const raw of names) {
        const name = canonicalName(raw);
        if (!name || isSeasoning(name) || wanted.includes(name)) {
          skipped.push(name || raw);
          continue;
        }
        wanted.push(name);
      }
      if (!wanted.length) return { added: [], skipped };
      const inserts = wanted.map((name) =>
        db
          .insert(s.pantryItems)
          .values({ id: newId(), groupId, name, addedOn: today })
          .onConflictDoNothing()
          .returning({ name: s.pantryItems.name }),
      );
      const results = await db.batch(
        inserts as [(typeof inserts)[number], ...(typeof inserts)[number][]],
      );
      const added: string[] = [];
      results.forEach((rows, i) => {
        if (rows.length) added.push(wanted[i]!);
        else skipped.push(wanted[i]!);
      });
      return { added, skipped };
    },
```
